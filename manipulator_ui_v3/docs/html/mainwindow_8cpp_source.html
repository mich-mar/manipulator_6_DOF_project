531 /**
532  * @brief Przetwarza nowe dane z czujników
533  *
534  * Funkcja aktualizuje:
535  * - Wykresy IMU (akcelerometr, żyroskop, magnetometr)
536  * - Wykresy kątów
537  * - Etykiety z wartościami
538  * - Wizualizacje potencjometrów
539  *
540  * Dodatkowo zarządza buforowaniem danych i usuwa stare punkty
541  * z wykresów gdy przekroczony zostanie limit.
542  *
543  * @param data Struktura zawierająca dane z czujników do przetworzenia
544  */
545 void MainWindow::processNewSensorData(const SensorDataPoint& data)
546 {
547  // Dodaj dane do kolejki
548  dataQueue.enqueue(data);
549  float time = dataCounter++;
550 
551  // Aktualizuj wykresy IMU
552  // Akcelerometr
553  accelSeriesX->append(time, data.imuData.accel.x);
554  accelSeriesY->append(time, data.imuData.accel.y);
555  accelSeriesZ->append(time, data.imuData.accel.z);
556 
557  // Żyroskop
558  gyroSeriesX->append(time, data.imuData.gyro.x);
559  gyroSeriesY->append(time, data.imuData.gyro.y);
560  gyroSeriesZ->append(time, data.imuData.gyro.z);
561 
562  // Magnetometr
563  magSeriesX->append(time, data.imuData.mag.x);
564  magSeriesY->append(time, data.imuData.mag.y);
565  magSeriesZ->append(time, data.imuData.mag.z);
566 
567  // Aktualizuj wykresy kątów (ADC)
568  for (int i = 0; i < 6; ++i) {
569  if (i < angleSeries.size()) {
570  // Konwersja napięcia na kąt (przykładowe mapowanie, dostosuj według potrzeb)
571  // Załóżmy, że 0V = -180°, 5V = 180°
572  double angle = (data.adcData.adc[i]);
573  angleSeries[i]->append(time, angle);
574 
575  // Usuń stare punkty jeśli przekroczono limit
576  if (angleSeries[i]->count() > MAX_POINTS) {
577  angleSeries[i]->remove(0);
578  }
579  }
580  }
581 
582  // Usuń stare punkty jeśli przekroczono limit dla wykresów IMU
583  if (accelSeriesX->count() > MAX_POINTS) {
584  // IMU series X
585  accelSeriesX->remove(0);
586  gyroSeriesX->remove(0);
587  magSeriesX->remove(0);
588 
589  // IMU series Y
590  accelSeriesY->remove(0);
591  gyroSeriesY->remove(0);
592  magSeriesY->remove(0);
593 
594  // IMU series Z
595  accelSeriesZ->remove(0);
596  gyroSeriesZ->remove(0);
597  magSeriesZ->remove(0);
598  }
599 
600  // Aktualizuj zakresy osi X
601  if (dataCounter > MAX_POINTS) {
602  float minX = dataCounter - MAX_POINTS;
603  float maxX = dataCounter;
604 
605  // IMU Charts
606  for (QChart* chart : {imuChartX, imuChartY, imuChartZ}) {
607  QValueAxis* axisX = qobject_cast<QValueAxis*>(chart->axes(Qt::Horizontal).first());
608  if (axisX) {
609  axisX->setRange(minX, maxX);
610  }
611  }
612 
613  // Angles Charts
614  for (QChart* chart : angleCharts) {
615  QValueAxis* axisX = qobject_cast<QValueAxis*>(chart->axes(Qt::Horizontal).first());
616  if (axisX) {
617  axisX->setRange(minX, maxX);
618  }
619  }
620  }
621 
622  // Aktualizuj wartości ADC w labelach
